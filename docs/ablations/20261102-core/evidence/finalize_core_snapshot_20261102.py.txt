def read(path):
    return json.loads(path.read_text())


def write(path, data):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(data, indent=2, ensure_ascii=False, allow_nan=False) + '\n')


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def copy(src, dst):
    dst.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(src, dst)
    assert sha(src) == sha(dst), str(src)


summary = read(RUN / 'summary.json')

test_rows = []
total_games = 0
for group in ('d4_distill_td', 'd4_scratch_td', 'plain_distill_td'):
    for seed in (0, 1, 2):
        path = RUN / group / f'seed{seed}'
        metrics = read(path / 'test/metrics.json')
        done = read(path / 'td/complete.json')
        assert metrics['params_sha256'] == done['best_params_sha256'] == frozen[f'{group}/{seed}']
        assert metrics['games'] == 2000 and metrics['seed_start'] == 500000
        assert not metrics['smoke'] and metrics['mode'] == 'greedy'
        assert metrics['source_sha256'] == experiment['source']['source_sha256']
        assert sha(path / 'test/games.csv') == metrics['csv_sha256']
        rows = list(csv.DictReader((path / 'test/games.csv').open()))
        assert [int(row['seed']) for row in rows] == list(range(500000, 502000))
        exponents = np.array([int(row['max_exponent']) for row in rows])
        assert float(exponents.mean()) == metrics['mean_max_exponent']
        for tile in (256, 512, 1024):
            rate = sum(int(row['max_tile']) >= tile for row in rows) / len(rows)
            assert abs(rate - metrics['reach_rates'][str(tile)]) < 1e-12
        total_games += len(rows)
        for name in ('metrics.json', 'games.csv'):
            copy(path / 'test' / name, DEST / 'results' / group / f'seed{seed}' / name)
        test_rows.append({
            'group': group, 'seed': seed, 'games': len(rows),
            'mean_max_exponent': metrics['mean_max_exponent'],
            'reach_256': metrics['reach_rates']['256'],
            'reach_512': metrics['reach_rates']['512'],
            'params_sha256': metrics['params_sha256'],
            'csv_sha256': metrics['csv_sha256'],
        })
